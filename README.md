-- Trigger cập nhật tồn kho sau khi thêm ImportDetail
DELIMITER //
CREATE TRIGGER trg_after_importdetail_insert
AFTER INSERT ON ImportDetail
FOR EACH ROW
BEGIN
    -- Chỉ chạy nếu Import.Status = 'Confirm'
    IF EXISTS (
        SELECT 1 FROM Import 
        WHERE ImportID = NEW.ImportID 
          AND Status = 'Confirm'
    ) THEN

        -- Nếu đã có dòng tồn kho thì cộng thêm
        IF EXISTS (
            SELECT 1 FROM Inventory
            WHERE ProductSizeID = NEW.ProductSizeID
              AND ExpiryDate = NEW.ExpiryDate
        ) THEN
            UPDATE Inventory
            SET ShelfQuantity = ShelfQuantity + NEW.Quantity
            WHERE ProductSizeID = NEW.ProductSizeID
              AND ExpiryDate = NEW.ExpiryDate;
        ELSE
            -- Nếu chưa có thì thêm mới
            INSERT INTO Inventory (ProductSizeID, ExpiryDate, ShelfQuantity)
            VALUES (NEW.ProductSizeID, NEW.ExpiryDate, NEW.Quantity);
        END IF;

    END IF;
END;

DELIMITER ;

CREATE TRIGGER trg_after_import_update
AFTER UPDATE ON Import
FOR EACH ROW
BEGIN
    -- Chỉ chạy khi status đổi từ Pending sang Confirm
    IF OLD.Status <> 'Confirm' AND NEW.Status = 'Confirm' THEN
        -- Lặp qua tất cả ImportDetail của Import này
        INSERT INTO Inventory (ProductSizeID, ExpiryDate, ShelfQuantity)
        SELECT d.ProductSizeID, d.ExpiryDate, d.Quantity
        FROM ImportDetail d
        WHERE d.ImportID = NEW.ImportID
        ON DUPLICATE KEY UPDATE ShelfQuantity = ShelfQuantity + VALUES(ShelfQuantity);
    END IF;
END;

DELIMITER ;
DELIMITER //
CREATE TRIGGER trg_after_import_cancel
AFTER UPDATE ON Import
FOR EACH ROW
BEGIN
    -- Chỉ chạy khi status đổi từ Confirm sang Cancelled
    IF OLD.Status = 'Confirm' AND NEW.Status = 'Cancelled' THEN
        -- Trừ tồn kho theo tất cả ImportDetail của Import này
        UPDATE Inventory i
        JOIN ImportDetail d 
          ON i.ProductSizeID = d.ProductSizeID 
         AND i.ExpiryDate = d.ExpiryDate
        SET i.ShelfQuantity = i.ShelfQuantity - d.Quantity
        WHERE d.ImportID = NEW.ImportID;
    END IF;
END;

-- Kiểm tra tồn kho theo sản phẩm và hạn sử dụng
SELECT 
    ps.ProductSizeID,
    p.Name AS ProductName,
    s.Type AS SizeType,
    SUM(inv.ShelfQuantity) AS TotalShelfQuantity
FROM `Inventory` inv
JOIN `ProductSize` ps ON inv.`ProductSizeID` = ps.`ProductSizeID`
JOIN `Product` p ON ps.`ProductID` = p.`ProductID`
JOIN `Size` s ON ps.`SizeID` = s.`SizeID`
GROUP BY ps.`ProductSizeID`, p.`Name`, s.`Type`
ORDER BY p.`Name`;

-- Tìm các sản phẩm sắp hết hạn trong 2 tháng và có số lượng trên kệ dưới 20
SELECT 
    inv.`ProductSizeID`,
    p.`Name` AS `ProductName`,
    s.`Type` AS `SizeType`,
    inv.`ExpiryDate`,
    inv.`ShelfQuantity`
FROM `Inventory` inv
JOIN `ProductSize` ps ON inv.`ProductSizeID` = ps.`ProductSizeID`
JOIN `Product` p ON ps.`ProductID` = p.`ProductID`
JOIN `Size` s ON ps.`SizeID` = s.`SizeID`
WHERE inv.`ShelfQuantity` <= 20
  AND inv.`ExpiryDate` <= CURDATE() + INTERVAL 2 MONTH
ORDER BY inv.`ExpiryDate` ASC;
